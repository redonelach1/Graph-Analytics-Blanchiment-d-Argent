\section{Schémas DDL SQL des Tables Delta Lake}
\label{app:schemas}

\begin{lstlisting}[language=SQL, caption={Définitions DDL des tables \texttt{silver\_transactions} et \texttt{fraud\_alerts}}, label={lst:ddl_tables}]
-- Table Silver des transactions validees et dedupliquees
CREATE TABLE IF NOT EXISTS silver_transactions (
    transaction_id STRING NOT NULL, source_account STRING NOT NULL,
    target_account STRING NOT NULL, montant DOUBLE NOT NULL,
    devise STRING NOT NULL, type_operation STRING NOT NULL,
    event_timestamp TIMESTAMP NOT NULL, payload_hash STRING NOT NULL,
    ingest_timestamp TIMESTAMP NOT NULL, event_date DATE NOT NULL
) USING delta PARTITIONED BY (event_date);

-- Table Gold des alertes de boucles frauduleuses
CREATE TABLE IF NOT EXISTS fraud_alerts (
    alert_id STRING NOT NULL, cycle_id STRING NOT NULL,
    rule_id STRING NOT NULL, rule_version INT NOT NULL,
    risk_level STRING NOT NULL, reason_codes ARRAY<STRING> NOT NULL,
    account_ids ARRAY<STRING> NOT NULL, transaction_ids ARRAY<STRING> NOT NULL,
    total_amount DOUBLE NOT NULL, currency STRING NOT NULL,
    first_detected_at TIMESTAMP NOT NULL, event_window_start TIMESTAMP NOT NULL,
    event_window_end TIMESTAMP NOT NULL, evidence_json STRING NOT NULL, alert_date DATE NOT NULL
) USING delta PARTITIONED BY (risk_level, alert_date);
\end{lstlisting}

  app_name: "AML-GraphShield-Engine"

  utc_enforced: true

  bootstrap_servers: "localhost:9092"

\section{Guide d'Exécution des Tests et Benchmarks}
\label{app:tests}

\begin{lstlisting}[language=bash, caption={Commandes de reproduction des validations unifiées}, label={lst:cmd_exec}]
# 1. Verification de l'environnement et test de fumee (Smoke test)
python scripts/smoke_test.py

# 2. Execution de la suite complete des 22 tests d'acceptation (AT-01 a AT-22)
pytest tests/ -v -s

# 3. Lancement des benchmarks de charge sous 3 paliers (10, 100, 500 TPS)
python scripts/run_benchmarks.py --levels L1,L2,L3 --output docs/benchmark_report.md

# 4. Compilation et demarrage du cockpit forensique Next.js 14
cd dashboard && npm run build && npm run start
\end{lstlisting}
